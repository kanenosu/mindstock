# blkf.150 ティトゥレスクの去就（1936年8月、専用ツリー読込時）
# ---------------------------------------------------------------
country_event = {
	id = blkf.150
	title = blkf.150.t
	desc = blkf.150.d
	picture = GFX_report_event_generic_conference
	fire_only_once = yes

	trigger = {
		tag = ROM
		has_country_flag = rom_balkan_tree_loaded
		date > 1936.8.28
	}
	mean_time_to_happen = { days = 5 }

	option = {
		# 留任させ、連邦との協調外交を続ける
		name = blkf.150.a
		add_political_power = 25
		if = { limit = { country_exists = BLK } BLK = { add_opinion_modifier = { target = ROM modifier = BLK_balkan_rapprochement } } }
		if = { limit = { country_exists = FRA } FRA = { add_opinion_modifier = { target = ROM modifier = BLK_trade_mission } } }
		ai_chance = { factor = 50 modifier = { factor = 3 has_completed_focus = rom_blk_rapprochement } }
	}
	option = {
		# 更迭し、親西欧路線を修正する
		name = blkf.150.b
		add_political_power = 50
		add_popularity = { ideology = fascism popularity = 0.03 }
		if = { limit = { country_exists = BLK } BLK = { add_opinion_modifier = { target = ROM modifier = BLK_refused_proposal } } }
		ai_chance = { factor = 50 modifier = { factor = 3 has_completed_focus = rom_blk_fortress_dobruja } }
	}
}

# ---------------------------------------------------------------
# blkf.154 ドブロジャ協定（方針 rom_blk_dobruja_accord）
# ---------------------------------------------------------------
country_event = {
	id = blkf.154
	title = blkf.154.t
	desc = blkf.154.d
	picture = GFX_report_event_generic_sign_treaty1
	is_triggered_only = yes
	fire_only_once = yes

	option = {
		# 補償と引き換えに譲渡する
		name = blkf.154.a
		if = {
			limit = { owns_state = 77 }
			remove_state_core = 77
			77 = { add_core_of = BLK }
			BLK = { transfer_state = 77 }
		}
		add_political_power = 100
		random_owned_controlled_state = {
			prioritize = { 46 }
			add_extra_state_shared_building_slots = 1
			add_building_construction = { type = industrial_complex level = 2 instant_build = yes }
		}
		if = { limit = { NOT = { has_idea = ROM_balkan_partnership } } add_ideas = ROM_balkan_partnership }
		if = { limit = { has_idea = ROM_dobruja_vigilance } remove_ideas = ROM_dobruja_vigilance }
		BLK = {
			add_opinion_modifier = { target = ROM modifier = BLK_sold_dobruja }
			country_event = { id = blk.90 days = 1 }
		}
		ai_chance = { factor = 55 }
	}
	option = {
		# 国際監視下の住民投票を提案する
		name = blkf.154.b
		add_stability = 0.02
		country_event = { id = blkf.160 days = 45 }
		ai_chance = { factor = 45 }
	}
}
