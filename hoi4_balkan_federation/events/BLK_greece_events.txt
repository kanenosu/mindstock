# blkf.109 国境事件（警戒路線で反復）

	id = blkf.109
	title = blkf.109.t
	desc = blkf.109.d
	picture = GFX_report_event_generic_army

	trigger = {
		tag = GRE
		has_country_flag = gre_balkan_tree_loaded
		has_completed_focus = gre_blk_northern_fortifications
		country_exists = BLK
		NOT = { has_war_with = BLK }
		NOT = { is_in_faction_with = BLK }
		NOT = { has_country_flag = gre_border_incident_cooldown }
	}
	mean_time_to_happen = { days = 300 }

	immediate = {
		hidden_effect = { set_country_flag = { flag = gre_border_incident_cooldown value = 1 days = 240 } }
	}

	option = {
		# 抗議にとどめる
		name = blkf.109.a
		add_political_power = 10
		add_opinion_modifier = { target = BLK modifier = BLK_border_incident }
		ai_chance = { factor = 60 }
	}
	option = {
		# 国境部隊を増強する
		name = blkf.109.b
		add_war_support = 0.03
		add_political_power = -25
		BLK = { add_opinion_modifier = { target = GRE modifier = BLK_border_incident } }
		ai_chance = { factor = 40 }
	}
