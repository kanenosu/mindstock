# バルカン連邦 スクリプト化エフェクト

# 連邦成立処理（ゲーム開始時に on_startup から呼ばれる）
# history/states の上書きを適用していない環境でも、ユーゴ・ブルガリア・アルバニアを
# 連邦に統合し、軍を引き継ぐためのフォールバック。
blk_establish_federation = {
	set_global_flag = blk_federation_established

	# 旧三国の中核州を連邦の中核に
	every_state = {
		limit = {
			OR = {
				is_core_of = YUG
				is_core_of = BUL
				is_core_of = ALB
				is_owned_by = YUG
				is_owned_by = BUL
				is_owned_by = ALB
			}
		}
		add_core_of = BLK
	}

	if = {
		limit = { country_exists = YUG }
		BLK = { annex_country = { target = YUG transfer_troops = yes } }
	}
	if = {
		limit = { country_exists = BUL }
		BLK = { annex_country = { target = BUL transfer_troops = yes } }
	}
	if = {
		limit = { country_exists = ALB }
		BLK = { annex_country = { target = ALB transfer_troops = yes } }
	}

	BLK = {
		if = {
			limit = { owns_state = 107 }
			set_capital = { state = 107 }
		}
		# AI の路線を抽選（AI戦略プランで参照）
		if = {
			limit = { is_ai = yes }
			random_list = {
				35 = { set_country_flag = blk_ai_path_crown }
				25 = { set_country_flag = blk_ai_path_people }
				25 = { set_country_flag = blk_ai_path_officers }
				15 = { set_country_flag = blk_ai_path_red }
			}
		}
		country_event = { id = blk.1 days = 1 }
	}

	# 周辺国の反応イベント
	if = { limit = { country_exists = GRE } GRE = { country_event = { id = blkf.1 days = 4 } } }
	if = { limit = { country_exists = ROM } ROM = { country_event = { id = blkf.2 days = 5 } } }
	if = { limit = { country_exists = HUN } HUN = { country_event = { id = blkf.3 days = 6 } } }
	if = { limit = { country_exists = ITA } ITA = { country_event = { id = blkf.4 days = 3 } } }
	if = { limit = { country_exists = GER } GER = { country_event = { id = blkf.5 days = 8 } } }
	if = { limit = { country_exists = SOV } SOV = { country_event = { id = blkf.6 days = 10 } } }
	if = { limit = { country_exists = ENG } ENG = { country_event = { id = blkf.7 days = 12 } } }
	if = { limit = { country_exists = FRA } FRA = { country_event = { id = blkf.8 days = 12 } } }
	if = { limit = { country_exists = TUR } TUR = { country_event = { id = blkf.9 days = 7 } } }
}

# 連邦への加盟（相手国を連邦構成国＝従属国にする）
blk_admit_to_federation = {
	# THIS = 加盟する国、FROM/BLK = 連邦
	add_opinion_modifier = { target = BLK modifier = BLK_joined_federation }
	BLK = {
		puppet = PREV
		add_political_power = 50
		add_stability = 0.03
	}
}

# 連邦構成国の完全統合（従属国を併合し中核化）
blk_integrate_member = {
	# THIS = 連邦、PREV なし。対象は事前に event_target として保存しておく
	every_state = {
		limit = { is_owned_by = event_target:blk_integration_target }
		add_core_of = BLK
	}
	annex_country = { target = event_target:blk_integration_target transfer_troops = yes }
	add_stability = -0.05
	add_political_power = -50
}

# 大バルカン連邦の宣言
blk_form_greater_balkans = {
	set_cosmetic_tag = BLK_greater
	every_owned_state = {
		limit = {
			OR = {
				is_core_of = GRE
				is_core_of = ROM
				is_core_of = YUG
				is_core_of = BUL
				is_core_of = ALB
			}
		}
		add_core_of = BLK
	}
	add_stability = 0.10
	add_war_support = 0.05
	add_political_power = 150
	set_country_flag = blk_greater_balkans_formed
	every_country = { news_event = { id = blknews.5 hours = 6 } }
}

# 旧ユーゴ・ブルガリア・アルバニア領への要求の設定（イタリアに対する）
blk_claim_adriatic_lost_lands = {
	ITA = {
		every_owned_state = {
			limit = {
				OR = {
					is_claimed_by = YUG
					is_core_of = YUG
					state = 163	# ザラ（ダルマチア沿岸のイタリア飛び地）
				}
			}
			add_claim_by = BLK
		}
	}
}

# 統合計画の進捗確認（5つ完了で「未完の連邦」を軽減）
blk_check_integration_progress = {
	if = {
		limit = {
			has_country_flag = blk_integrated_croatia
			has_country_flag = blk_integrated_slovenia
			has_country_flag = blk_integrated_bulgaria
			has_country_flag = blk_integrated_macedonia
			has_country_flag = blk_integrated_albania
		}
		if = {
			limit = { has_idea = BLK_unfinished_federation }
			swap_ideas = { remove_idea = BLK_unfinished_federation add_idea = BLK_unfinished_federation_2 }
		}
		else_if = {
			limit = { has_idea = BLK_unfinished_federation_2 }
			remove_ideas = BLK_unfinished_federation_2
		}
		add_political_power = 100
		add_stability = 0.05
	}
}
