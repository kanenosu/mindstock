# 連邦に加盟する余地のある隣国か

	OR = {
		original_tag = GRE
		original_tag = ROM
		original_tag = TUR
	}
	is_subject = no
	NOT = { has_war_with = BLK }
}

# 連邦が「権威主義軍事路線」にあるか

		has_completed_focus = blk_officers_federation
