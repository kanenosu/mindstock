# 連邦構成国：連邦に加盟した国が得る自治レベル（従属国だが高い自治を持つ）
autonomy_state = {
	id = autonomy_blk_federal_member
	default = no
	is_puppet = yes
	use_overlord_color = yes
	min_freedom_level = 0.6
	manpower_influence = 0.25

	rule = {
		can_not_declare_war = yes
		can_decline_call_to_war = no
		can_create_factions = no
		can_join_factions = no
		can_send_volunteers = yes
		can_guarantee_other_ideologies = no
		can_not_build_buildings_in_overlord = no
	}

	modifier = {
		autonomy_gain = 0.2
		autonomy_manpower_share = 0.25
		cic_to_overlord_factor = 0.15
		mic_to_overlord_factor = 0.15
		extra_trade_to_overlord_factor = 0.25
		overlord_trade_cost_factor = -0.5
		license_subject_master_purchase_cost = -0.5
		can_master_build_for_overlord = yes
		stability_factor = 0.05
	}

	ai_subject_wants_higher = {
		factor = 0.5
	}
	ai_overlord_wants_lower = {
		factor = 0.5
	}

	allowed = {
		OVERLORD = { original_tag = BLK }
	}

	can_take_level = {
		OR = {
			OVERLORD = { has_government = democratic }
			has_opinion = { target = OVERLORD value > 80 }
		}
	}
	can_lose_level = {
		OR = {
			OVERLORD = { has_government = fascism }
			has_war_with = OVERLORD
			has_opinion = { target = OVERLORD value < 0 }
		}
	}
}
